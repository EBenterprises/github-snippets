        const basePicker = document.getElementById('basePicker');
        const mudguardPicker = document.getElementById('mudguardPicker');
        const swooshPicker = document.getElementById('swooshPicker');
        const solePicker = document.getElementById('solePicker');

        const shoeBase = document.getElementById('shoe-base');
        const shoeMudguard = document.getElementById('shoe-mudguard');
        const shoeSwoosh = document.getElementById('shoe-swoosh');
        const shoeSole = document.getElementById('shoe-sole');

        basePicker.addEventListener('input', (e) => shoeBase.style.fill = e.target.value);
        mudguardPicker.addEventListener('input', (e) => shoeMudguard.style.fill = e.target.value);
        swooshPicker.addEventListener('input', (e) => shoeSwoosh.style.fill = e.target.value);
        solePicker.addEventListener('input', (e) => shoeSole.style.fill = e.target.value);

        async function saveDesign() {
            const response = await fetch('/api/save-design', {
                method: 'POST',
                headers: { 'Content-Type': 'application/json' },
                body: JSON.stringify({ 
                    designName: 'My Custom Sneaker', 
                    colors: { 
                        base: basePicker.value, 
                        mudguard: mudguardPicker.value,
                        swoosh: swooshPicker.value, 
                        sole: solePicker.value 
                    } 
                })
            });
            const result = await response.json();
            alert(result.message);
        }
    </script>
