cat << 'EOF' > new_module.sh
#!/usr/bin/env bash
# ==============================================================================
# Executive Spectrum - Module Scaffolding Tool
# EB Enterprises | Automated Terminal Environment
# ==============================================================================

MODULE_NAME="$1"

if [[ -z "$MODULE_NAME" ]]; then
    echo "Usage: $0 <module_name>"
    exit 1
fi

mkdir -p "$MODULE_NAME"/{backend,frontend,assets,docs}

cat << 'INNER_EOF' > "$MODULE_NAME/README.md"
# $MODULE_NAME
Executive Spectrum Modular Component | EB Enterprises
INNER_EOF

cat << 'INNER_EOF' > "$MODULE_NAME/backend/server.py"
import os
import http.server
import socketserver

PORT = 8080
Handler = http.server.SimpleHTTPRequestHandler

with socketserver.TCPServer(("", PORT), Handler) as httpd:
    print(f"Serving module on port {PORT}...")
    httpd.serve_forever()
INNER_EOF

chmod +x "$MODULE_NAME/backend/server.py"
echo "Successfully scaffolded module: $MODULE_NAME"
EOF

chmod +x new_module.sh
