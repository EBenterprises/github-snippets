                headers: { 'Content-Type': 'application/json' },
                body: JSON.stringify({ 
                    designName: 'Custom Air Silhouette', 
                    colors: { 
                        base: basePicker.value, 
                        mudguard: mudguardPicker.value,
                        swoosh: swooshPicker.value, 
                        eyestay: eyestayPicker.value,
