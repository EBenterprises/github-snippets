const express = require('express');
const fs = require('fs');
const path = require('path');
const app = express();
const PORT = 3002;

app.use(express.json());
app.use(express.static('public'));

const DATA_FILE = path.join(__dirname, 'data', 'designs.json');
const USERS_FILE = path.join(__dirname, 'data', 'users.json');

if (!fs.existsSync(path.dirname(DATA_FILE))) {
    fs.mkdirSync(path.dirname(DATA_FILE), { recursive: true });
}

const readData = (file) => fs.existsSync(file) ? JSON.parse(fs.readFileSync(file)) : [];
const writeData = (file, data) => fs.writeFileSync(file, JSON.stringify(data, null, 2));

app.post('/api/register', (req, res) => {
    const { username, email, password } = req.body;
    const users = readData(USERS_FILE);

    if (users.find(u => u.email === email)) {
        return res.status(400).json({ success: false, message: 'User already exists.' });
    }

    const newUser = { id: Date.now(), username, email, password };
    users.push(newUser);
    writeData(USERS_FILE, users);

    res.json({ success: true, message: 'Account created successfully!', userId: newUser.id });
});

app.post('/api/save-design', (req, res) => {
    const { userId, designName, colors, materials } = req.body;
    const designs = readData(DATA_FILE);

    const newDesign = {
        designId: Date.now(),
        userId: userId || 'anonymous',
        designName: designName || 'Untitled Custom',
        colors,
        materials,
        createdAt: new Date().toISOString()
    };

    designs.push(newDesign);
    writeData(DATA_FILE, designs);

    res.json({ success: true, message: 'Sneaker design saved successfully!', designId: newDesign.designId });
});

app.get('/api/designs', (req, res) => {
    const designs = readData(DATA_FILE);
    res.json({ success: true, designs });
});

app.listen(PORT, () => {
    console.log(`Server running at http://localhost:${PORT}`);
});
