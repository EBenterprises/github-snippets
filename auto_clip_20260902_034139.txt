cat << 'EOF' >> ~/.bashrc

# --- Automated GitHub Snippet & Repo Sync Anchor ---
auto_sync_snippets() {
    local target="$HOME/github-snippets"
    if [ -d "$target" ]; then
        cd "$target" || return
        git add -A >/dev/null 2>&1
        if ! git diff-index --quiet HEAD -- 2>/dev/null; then
            git commit -m "Auto-sync snippet update: $(date '+%Y-%m-%d %H:%M:%S')" >/dev/null 2>&1
            git push origin main --force-with-lease >/dev/null 2>&1 &
        fi
        cd - >/dev/null 2>&1
    fi
}

# Trigger sync quietly in the background on shell startup
auto_sync_snippets &
EOF

# Apply immediately to current session
source ~/.bashrc
echo "Snippet sync anchor successfully embedded into shell profile."
