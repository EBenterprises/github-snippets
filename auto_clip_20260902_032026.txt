    <script>
        const renderStage = document.getElementById('renderStage');
        const zoomRange = document.getElementById('zoomRange');
        const shoeColor = document.getElementById('shoeColor');
        const tintBox = document.getElementById('tintBox');
        const textInput = document.getElementById('textInput');
        const customTextOverlay = document.getElementById('customTextOverlay');

        zoomRange.addEventListener('input', (e) => {
            renderStage.style.transform = `scale(${e.target.value})`;
        });

        shoeColor.addEventListener('input', (e) => {
            tintBox.style.backgroundColor = e.target.value;
        });

        textInput.addEventListener('input', (e) => {
            customTextOverlay.textContent = e.target.value || '';
        });

        async function saveDesign() {
            const res = await fetch('/api/save-design', {
                method: 'POST',
                headers: { 'Content-Type': 'application/json' },
                body: JSON.stringify({ 
                    designName: textInput.value || 'Custom Kick', 
                    color: shoeColor.value,
                    zoom: zoomRange.value 
                })
            });
            const data = await res.json();
            alert(data.message);
        }
    </script>
